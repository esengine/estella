import { Tabs, TabItem, Steps, Aside } from '@astrojs/starlight/components';

<Tabs>
  <TabItem label="Use the editor">
    <Steps>

    1. Download the latest editor — from the
       [mirror](https://dl.estella.games/latest.json) if GitHub is slow where you
       are, or from the
       [releases page](https://github.com/esengine/estella/releases), which is
       where every release is published.

    2. Install and open it, then click **New project**.

    3. Pick a template — the **Blank** starter, or one of the bundled
       **Examples** — enter a project name and location, and click
       **Create project**. With Blank, the editor opens a default scene with a
       Camera entity.

    </Steps>

    <Aside type="tip">
      **Help → Check for Updates…** reads the same mirror (`dl.estella.games`) and
      hands back a download from it, falling back to GitHub when the mirror has no
      answer or a staler one — the origin stays the authority on what exists.
    </Aside>

    <Aside type="note">
      Installers are currently unsigned, so on first launch Windows SmartScreen
      or macOS Gatekeeper may warn about an unknown developer — choose **Run
      anyway** / allow it in **System Settings → Privacy & Security**.
    </Aside>
  </TabItem>

  <TabItem label="Build from source">
    Prerequisites:

    - **Node.js** ≥ 18 and **pnpm** ≥ 8
    - **Emscripten** 5.0.0 (exact version, for CI parity)

    <Steps>

    1. Clone (with submodules) and install:

       ```bash
       git clone --recursive https://github.com/esengine/estella.git
       cd estella
       pnpm install
       pnpm emsdk:setup   # one-time: installs Emscripten 5.0.0 into tools/emsdk
       ```

    2. Build the engine, SDK, and sync to the editor:

       ```bash
       node build-tools/cli.js build -t all
       ```

    3. Run the editor:

       ```bash
       cd desktop && pnpm dev
       ```

    </Steps>

    See [CONTRIBUTING.md](https://github.com/esengine/estella/blob/master/CONTRIBUTING.md)
    for the full build reference.
  </TabItem>
</Tabs>
