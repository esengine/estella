import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Fast" icon="rocket">
    A C++ rendering pipeline compiled to WebAssembly — not interpreted JavaScript.
  </Card>
  <Card title="Type-safe SDK" icon="seti:typescript">
    A first-class TypeScript API: `defineComponent`, `defineSystem`, `Query`,
    `Commands`, `Res`.
  </Card>
  <Card title="Data-oriented" icon="puzzle">
    An Entity-Component-System core with a single cache-friendly storage layer.
  </Card>
  <Card title="Visual editor" icon="pencil">
    A UE5-style editor — scene viewport, inspector, tilemap mode, Sequencer, and
    node-graph editors for materials and AI.
  </Card>
  <Card title="Batteries included" icon="star">
    2D and 3D physics, glTF/FBX models with PBR materials and lights, terrain,
    tilemaps, particles, 2D lighting, animation, Spine, audio, UI, and a gameplay AI layer.
  </Card>
  <Card title="Cross-platform" icon="laptop">
    One project, exported to the web, desktop, WeChat MiniGames, single-file
    playable ads, and native Android / iOS apps.
  </Card>
  <Card title="Agent-native" icon="comment">
    The editor is an MCP server, and it ships an agent of its own. Both drive the
    same hundred-odd tools the UI does — so an agent can build a scene, press play,
    and look at a screenshot of what it made.
  </Card>
</CardGrid>
